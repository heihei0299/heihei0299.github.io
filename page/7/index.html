server_name _; # 匹配任意 Host include /etc/nginx/conf.d/cf-ip; deny all; return 404; } vim /etc/nginx/conf.d/typecho.conf server { listen 80; server_name yourservername; root /var/www/html/typecho; index index.html index.htm index.php;
